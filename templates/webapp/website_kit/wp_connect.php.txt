function bookpilot_connect_defaults() {
    return array(
        'base' => '{{ base }}',
        'kit_id' => '{{ kit_id }}',
        'booking_slug' => '{{ booking_slug }}',
        'careers_slug' => '{{ careers_slug }}',
        'color' => '{{ color }}',
    );
}

function bookpilot_connect_opt($key) {
    $saved = get_option('bookpilot_connect', array());
    $defaults = bookpilot_connect_defaults();
    return isset($saved[$key]) && $saved[$key] !== '' ? $saved[$key] : (isset($defaults[$key]) ? $defaults[$key] : '');
}

function bookpilot_connect_url($path) {
    return rtrim(bookpilot_connect_opt('base'), '/') . $path;
}

function bookpilot_connect_attrs($atts) {
    $out = '';
    foreach ($atts as $key => $value) {
        if ($value !== '' && $value !== null) {
            $out .= ' data-' . sanitize_key($key) . '="' . esc_attr($value) . '"';
        }
    }
    return $out;
}

function bookpilot_connect_widget($class, $script_handle, $path, $atts) {
    wp_enqueue_script($script_handle, esc_url(bookpilot_connect_url($path)), array(), '1.0.0', true);
    return '<div class="' . esc_attr($class) . '"' . bookpilot_connect_attrs($atts) . '></div>';
}

function bookpilot_connect_fetch($path, $cache_key) {
    $cached = get_transient($cache_key);
    if ($cached !== false) {
        return $cached;
    }
    $response = wp_remote_get(bookpilot_connect_url($path), array('timeout' => 10));
    if (is_wp_error($response) || wp_remote_retrieve_response_code($response) !== 200) {
        return null;
    }
    $data = json_decode(wp_remote_retrieve_body($response), true);
    if (!is_array($data)) {
        return null;
    }
    set_transient($cache_key, $data, 5 * MINUTE_IN_SECONDS);
    return $data;
}

function bookpilot_connect_item_html($item, $data, $units) {
    $out = !isset($item['available']) || $item['available'];
    $html = '<div class="bookpilot-item' . ($out ? '' : ' sold-out') . '">';
    if (!empty($item['image'])) {
        $html .= '<img src="' . esc_url($item['image']) . '" alt="' . esc_attr($item['name']) . '" loading="lazy">';
    }
    $html .= '<div class="bp-body"><strong>' . esc_html($item['name']) . '</strong>';
    if (!empty($item['vegetarian'])) { $html .= '<span class="bookpilot-tag veg">Veg</span>'; }
    if (!empty($item['featured'])) { $html .= '<span class="bookpilot-tag hot">★</span>'; }
    if (!$out) { $html .= '<span class="bookpilot-tag out">Sold out</span>'; }
    if (!empty($item['location'])) { $html .= '<div class="bp-desc">' . esc_html($item['location']) . '</div>'; }
    if (!empty($item['description'])) { $html .= '<div class="bp-desc">' . esc_html($item['description']) . '</div>'; }
    if (!empty($item['price'])) {
        $unit = isset($item['unit'], $units[$item['unit']]) ? ' ' . $units[$item['unit']] : '';
        $html .= '<div class="bp-price">' . esc_html($data['currency'] . ' ' . $item['price'] . $unit) . '</div>';
    }
    return $html . '</div></div>';
}

function bookpilot_connect_catalogue($atts) {
    $a = shortcode_atts(array('category' => '', 'limit' => 0, 'group' => 'yes', 'hide_sold_out' => 'no'), $atts, 'bookpilot_catalogue');
    $data = bookpilot_connect_fetch('/kit/' . rawurlencode(bookpilot_connect_opt('kit_id')) . '/catalogue.json', 'bookpilot_catalogue_' . md5(bookpilot_connect_opt('kit_id')));
    if (!$data || empty($data['items'])) {
        return '';
    }
    $units = array('hour' => '/ hour', 'day' => '/ day', 'night' => '/ night', 'month' => '/ month', 'monthly' => '/ month');
    $groups = array();
    $count = 0;
    foreach ($data['items'] as $item) {
        if ($a['category'] !== '' && (!isset($item['category']) || strcasecmp($item['category'], $a['category']) !== 0)) {
            continue;
        }
        if ($a['hide_sold_out'] === 'yes' && isset($item['available']) && !$item['available']) {
            continue;
        }
        if ((int) $a['limit'] > 0 && $count >= (int) $a['limit']) {
            break;
        }
        $count++;
        $key = ($a['group'] === 'yes' && $a['category'] === '' && !empty($item['category'])) ? $item['category'] : '';
        $groups[$key][] = $item;
    }
    $html = bookpilot_connect_styles() . '<div class="bookpilot-menu">';
    if (count($groups) > 1) {
        $html .= '<nav class="bookpilot-tabs">';
        foreach (array_keys($groups) as $i => $name) {
            if ($name !== '') { $html .= '<a href="#bp-cat-' . $i . '">' . esc_html($name) . '</a>'; }
        }
        $html .= '</nav>';
    }
    $i = 0;
    foreach ($groups as $name => $items) {
        if ($name !== '') { $html .= '<h3 class="bookpilot-cat" id="bp-cat-' . $i . '">' . esc_html($name) . '</h3>'; }
        $html .= '<div class="bookpilot-grid">';
        foreach ($items as $item) { $html .= bookpilot_connect_item_html($item, $data, $units); }
        $html .= '</div>';
        $i++;
    }
    return $html . '</div>';
}
add_shortcode('bookpilot_catalogue', 'bookpilot_connect_catalogue');

/* [bookpilot_offers] offers running today from the BookPilot POS (Pricing → Offers) */
function bookpilot_connect_offers($atts) {
    $data = bookpilot_connect_fetch('/kit/' . rawurlencode(bookpilot_connect_opt('kit_id')) . '/offers.json', 'bookpilot_offers_' . md5(bookpilot_connect_opt('kit_id')));
    if (!$data || empty($data['items'])) {
        return '';
    }
    $html = bookpilot_connect_styles() . '<div class="bookpilot-offers">';
    foreach ($data['items'] as $offer) {
        $html .= '<div class="bookpilot-offer"><div class="bp-big">' . esc_html($offer['discount']) . ' off</div><strong>' . esc_html($offer['name']) . '</strong>';
        if (!empty($offer['product'])) { $html .= '<div>' . esc_html($offer['product']) . '</div>'; }
        $html .= '<div style="opacity:.7;font-size:13px">Until ' . esc_html(date_i18n(get_option('date_format'), strtotime($offer['until']))) . '</div></div>';
    }
    return $html . '</div>';
}
add_shortcode('bookpilot_offers', 'bookpilot_connect_offers');

function bookpilot_connect_register() {
    register_setting('bookpilot_connect', 'bookpilot_connect', array('type' => 'array', 'sanitize_callback' => 'bookpilot_connect_sanitize'));
}

function bookpilot_connect_sanitize($input) {
    $clean = array();
    $clean['base'] = isset($input['base']) ? esc_url_raw($input['base']) : '';
    foreach (array('kit_id', 'booking_slug', 'careers_slug') as $key) {
        $clean[$key] = isset($input[$key]) ? sanitize_title($input[$key]) : '';
    }
    delete_transient('bookpilot_catalogue_' . md5(bookpilot_connect_opt('kit_id')));
    delete_transient('bookpilot_info_' . md5(bookpilot_connect_opt('kit_id')));
    delete_transient('bookpilot_offers_' . md5(bookpilot_connect_opt('kit_id')));
    return $clean;
}
